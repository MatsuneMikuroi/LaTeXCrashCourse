\begin{frame}[t]{Writting}
  \onslide<+->
  2 types of "text" boxes:
  \begin{itemize}
    \item Text box: for textual content (no delimiter)
    \item Math box: for mathematical content (delimiter: \(\backslash(...\backslash)\) or \(\$...\$\) for inline, \(\backslash[...\backslash]\) or \(\$\$...\$\$\) for display)
  \end{itemize}
  \onslide<+->
  2 types of macros:
  \begin{itemize}
    \item Command: inline content, e.g. \texttt{\(\backslash\)LaTeX}
    \item Environment: block of content with a begin and an end, e.g. \texttt{\(\backslash\)begin\{itemize\}...\(\backslash\)end\{itemize\}}
  \end{itemize}
\end{frame}

\begin{frame}[t]{Preamble}
  \onslide<+->
  Before \texttt{\(\backslash\)begin\{document\}}:
  \begin{itemize}
    \item Document class and options, \texttt{\(\backslash\)documentclass\{...\}}
    \item Packages and their options, \texttt{\(\backslash\)usepackage\{...\}}
    \item Custom commands and environments
    \item Document metadata, \texttt{\(\backslash\)title\{...\}}, \texttt{\(\backslash\)author\{...\}}, \texttt{\(\backslash\)date\{...\}}
  \end{itemize}
\end{frame}

\begin{frame}[t]{Document management}
  \onslide<+->
  Use the sectioning commands to structure your document:
  \begin{itemize}
    \item \texttt{\(\backslash\)part\{...\}} for parts
    \item \texttt{\(\backslash\)chapter\{...\}} for chapters (only in book and report document classes)
    \item \texttt{\(\backslash\)section\{...\}} for sections
    \item \texttt{\(\backslash\)subsection\{...\}} for subsections
    \item \texttt{\(\backslash\)subsubsection\{...\}} for subsubsections
    \item \texttt{\(\backslash\)paragraph\{...\}} for paragraphs
    \item \texttt{\(\backslash\)subparagraph\{...\}} for subparagraphs
  \end{itemize}
  \onslide<+->
  Create a TOC with the \texttt{\(\backslash\)tableofcontents}, \texttt{\(\backslash\)listoffigures} and \texttt{\(\backslash\)listoftables} commands (depth: \texttt{\(\backslash\)setcounter\{tocdepth\}\{...\}})
\end{frame}

\begin{frame}[t]{Document management (cont.)}
  \onslide<+->
  Reference:
  \begin{itemize}
    \item \texttt{\(\backslash\)label\{...\}} command to label sections, figures, tables, equations, etc.
    \item \texttt{\(\backslash\)ref\{...\}} command to reference the labeled items in the text
    \item \texttt{\(\backslash\)pageref\{...\}} command to reference the page number of the labeled item
  \end{itemize}
  \onslide<+->
  Multiple files management:
  \begin{itemize}
    \item Use the \texttt{\(\backslash\)include\{...\}} command to include other .tex files in the main document
    \item Use the \texttt{\(\backslash\)input\{...\}} command to include other .tex files without starting a new page
  \end{itemize}
\end{frame}

  \onslide<+->
  Just write your text, and use command to format it:
  \begin{itemize}
    \item \textbf{Bold} with \texttt{\(\backslash\)textbf\{...\}}
    \item \textit{Italic} with \texttt{\(\backslash\)textit\{...\}}
    \item \underline{Underline} with \texttt{\(\backslash\)underline\{...\}}
    \item {\color{red} Colored} with \texttt{\(\backslash\)color\{red\}...}
  \end{itemize}

  You can also use some environments:
  \begin{itemize}
    \item Itemize for unordered list
    \item Enumerate for ordered list
    \item Description for description list
    \item And many more...
  \end{itemize}
\end{frame}

\begin{frame}[t]{Math box}
  \onslide<+->
  Special characters and symbols:
  \begin{itemize}
    \item Greek letters: \texttt{\(\backslash\)alpha}, \texttt{\(\backslash\)beta}, ...
    \item Operators: \texttt{\(\backslash\)sum}, \texttt{\(\backslash\)int}, \texttt{\(\backslash\)frac\{...\}\{...\}}, ...
    \item Relations: \texttt{\(\backslash\)leq}, \texttt{\(\backslash\)geq}, ...
    \item And many more...
  \end{itemize}
  \onslide<+->
  Some maths environments:
  \begin{itemize}
    \item Equation for numbered equations
    \item Align for aligned equations
    \item And many more...
  \end{itemize}
  For numbered maths environments, you can add a * to the end of the environment name to remove the numbering
\end{frame}

\begin{frame}[t]{Figures and tables}
  \onslide<+->
  Use the figure environment to include images:
  \begin{itemize}
    \item \texttt{\(\backslash\)includegraphics\{...\}} to include the image
    \item \texttt{\(\backslash\)caption\{...\}} to add a caption
    \item \texttt{\(\backslash\)label\{...\}} to add a label for referencing
  \end{itemize}
  \onslide<+->
  Use the table environment to create tables:
  \begin{itemize}
    \item Use the tabular environment to define the structure of the table
    \item Use \(\&\) to separate columns and \(\backslash\backslash\) to separate rows
    \item Use \texttt{\(\backslash\)hline} to add horizontal lines
    \item Use \texttt{\(\backslash\)caption\{...\}} and \texttt{\(\backslash\)label\{...\}} for caption and referencing
  \end{itemize}
  array act as a math version of tabular
\end{frame}

\begin{frame}[t]{Bibliography}
  \onslide<+->
  Bibliography environment to create a bibliography:
  \begin{itemize}
    \item Use the \texttt{\(\backslash\)bibitem\{...\}} command to add a reference
    \item Use the \texttt{\(\backslash\)cite\{...\}} command to cite a reference in the text
  \end{itemize}
  \onslide<+->
  BibTeX or BibLaTeX for advanced management
\end{frame}

\begin{frame}[t]{Create figure and graphes}
  \onslide<+->
  TikZ package to create figures and graphes directly in LaTeX:
  \begin{itemize}
    \item \texttt{tikzpicture} environment to create a drawing area
    \item Use various TikZ commands to draw shapes, lines, and add text
    \item TikZ allows for complex and customizable figures, including flowcharts, diagrams, and more
  \end{itemize}
  \onslide<+->
  One of most complete packages, look the doc or search online
\end{frame}

\begin{frame}[t]{Create your own commands and environments}
  \onslide<+->
  \texttt{\(\backslash\)newcommand\{...\}\{...\}} command:
  \begin{itemize}
    \item Define a new command name and its definition
    \item Can take arguments for more flexibility
  \end{itemize}
  \onslide<+->
  \texttt{\(\backslash\)newenvironment\{...\}\{...\}\{...\}} command:
  \begin{itemize}
    \item Define a new environment name, its begin code, and its end code
    \item Can also take arguments for more flexibility
  \end{itemize}
  \onslide<+->
  For argument, use [X] for optional argument (in order) and \(\#\)X+n for mandatory argument (X+n is the number of the argument)
\end{frame}

\begin{frame}[t]{Issue with macros definition}
  \onslide<+->
  \begin{itemize}
    \item Hard to follow argument if many
    \item Hard to read the definition of the command or environment
    \item Hard to input because of the syntax
  \end{itemize}
  \onslide<+->
  \texttt{xparse} package to create more user-friendly commands and environments:
  \begin{itemize}
    \item \texttt{\(\backslash\)NewDocumentCommand\{name\}\{arguments\}\{definition\}}
    \item \texttt{\(\backslash\)NewDocumentEnvironment\{name\}\{arguments\}\{begin code\}\{end code\}}
  \end{itemize}
\end{frame}

\end{frame}
