\begin{frame}[t]{Writting}
  \onslide<+->
  2 types of "text" boxes:
  \begin{itemize}
    \item Text box: no delimiter
    \item Math box:
    \begin{itemize}
      \item Inline: \(\backslash(...\backslash)\) or \(\$...\$\)
      \item Display: \(\backslash[...\backslash]\) or \(\$\$...\$\$\)
    \end{itemize}
  \end{itemize}
  \onslide<+->
  2 types of macros:
  \begin{itemize}
    \item Command: \texttt{\(\backslash\)LaTeX}
    \item Environment: \texttt{\(\backslash\)begin\{itemize\}...\(\backslash\)end\{itemize\}}
  \end{itemize}
  \note{Command: inline content. Environment: block of content with a begin and an end.}
\end{frame}

\begin{frame}[t]{When it does not compile}
  \onslide<+->
  Logs \(\rightarrow\) \textbf{first} error first:
  \begin{itemize}
    \item Undefined control sequence: typo, missing package
    \item Missing \$ inserted: math outside math mode
    \item File ended while scanning use of...: missing \}
    \item Environment ... undefined: missing package
    \item File ... not found: file name, case sensitive
  \end{itemize}
  \onslide<+->
  PDF despite errors \(\Rightarrow\) maybe wrong
  \note{Fix the first error first, the others are often consequences. The line of the error is given as l.12 in the log. Overleaf compiles despite errors: the PDF may be wrong, never ignore an error. If Overleaf acts strangely: Recompile from scratch (arrow next to Recompile).}
\end{frame}

\begin{frame}[t]{Create your own commands and environments}
  \onslide<+->
  \texttt{\(\backslash\)newcommand\{...\}\{...\}}:
  \begin{itemize}
    \item Name + definition
    \item Arguments
  \end{itemize}
  \onslide<+->
  \texttt{\(\backslash\)newenvironment\{...\}\{...\}\{...\}}:
  \begin{itemize}
    \item Name + begin code + end code
    \item Arguments
  \end{itemize}
  \onslide<+->
  Arguments:
  \begin{itemize}
    \item \texttt{[n]}: number of arguments
    \item \texttt{[default]}: optional first argument
    \item \texttt{\#1}, \texttt{\#2}...: in the definition
  \end{itemize}
  \note{Example: \textbackslash newcommand\{\textbackslash vect\}[1]\{\textbackslash mathbf\{\#1\}\}, then \textbackslash vect\{x\}.}
\end{frame}

\begin{frame}[t]{Issue with macros definition}
  \onslide<+->
  \begin{itemize}
    \item Many arguments: hard to follow
    \item Definition: hard to read
    \item Syntax: hard to input
  \end{itemize}
  \onslide<+->
  \texttt{xparse} package:
  \begin{itemize}
    \item \texttt{\(\backslash\)NewDocumentCommand\{name\}\{arguments\}\{definition\}}
    \item \texttt{\(\backslash\)NewDocumentEnvironment\{name\}\{arguments\}\{begin code\}\{end code\}}
  \end{itemize}
  \note{xparse creates more user-friendly commands and environments. Since 2020 these commands are built into LaTeX, no package needed.}
\end{frame}
